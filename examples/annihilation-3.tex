\section*{Annihilation 3}

From the first section we have
\begin{align*}
\sum_{ab}\mathcal M_1{}^{\mu\nu}\mathcal M^*_{1\mu\nu}
&=\frac{e^4}{(t-m^2)^2}
\left(8 t u - 24 m^2 t - 8 m^2 u - 8 m^4\right)
\\
\sum_{ab}\mathcal M_1{}^{\mu\nu}\mathcal M^*_{2\nu\mu}
&=\frac{e^4}{(t-m^2)(u-m^2)}
\left(8 m^2 s - 32 m^4\right)
\\
\sum_{ab}\mathcal M_2{}^{\nu\mu}\mathcal M^*_{2\nu\mu}
&=\frac{e^4}{(u-m^2)^2}
\left(8 t u - 8 m^2 t - 24 m^2 u - 8 m^4\right)
\end{align*}

We also have by computer algebra
\begin{align*}
\sum_{ab}\mathcal M_1{}^{\mu\nu}\mathcal M^*_{1\mu\nu}
&=\frac{e^4}{(t-m^2)^2}
\left(
32 (p_1\cdot p_3) (p_1\cdot p_4) + 32 m^2 (p_1\cdot p_3) - 32 m^4
\right)
\\
\sum_{ab}\mathcal M_1{}^{\mu\nu}\mathcal M^*_{2\nu\mu}
&=\frac{e^4}{(t-m^2)(u-m^2)}
\left(
16 m^2 (p_1\cdot p_2) - 16 m^4
\right)
\\
\sum_{ab}\mathcal M_2{}^{\nu\mu}\mathcal M^*_{2\nu\mu}
&=\frac{e^4}{(u-m^2)^2}
\left(
32 (p_1\cdot p_3) (p_1\cdot p_4) + 32 m^2 (p_1\cdot p_4) - 32 m^4
\right)
\end{align*}
