"Bloch sphere"

chi = (A exp(i phi1), sqrt(1 - A^2) exp(i phi2))

check(dot(conj(chi),chi) == 1)

s = exp(-i phi1) chi

check(dot(conj(s),s) == 1)

Sx = 1/2 hbar ((0,1),(1,0))
Sy = 1/2 hbar ((0,-i),(i,0))
Sz = 1/2 hbar ((1,0),(0,-1))

check(dot(conj(chi),Sx,chi) == dot(conj(s),Sx,s))
check(dot(conj(chi),Sy,chi) == dot(conj(s),Sy,s))
check(dot(conj(chi),Sz,chi) == dot(conj(s),Sz,s))

"ok"
