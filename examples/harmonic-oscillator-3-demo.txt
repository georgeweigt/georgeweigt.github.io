"Harmonic oscillator 3"

aplus = sqrt(hbar / (2 m omega)) (m omega x / hbar - i p / hbar)
aminus = sqrt(hbar / (2 m omega)) (m omega x / hbar + i p / hbar)

"Verify equation (1)"
check(x == sqrt(hbar / (2 m omega)) (aplus + aminus))
"ok"

"Verify equation (2)"
check(p == i sqrt(m hbar omega / 2) (aplus - aminus))
"ok"
