\section*{Free particle propagator 1}

Let $K(b,a)$ be the amplitude for a particle to go from $a$ to $b$
where $\hat H$ is the Hamiltonian.
\begin{equation*}
K(b,a)=\langle x_b|\exp\left(-\frac{i}{\hbar}(t_b-t_a)\hat H\right)|x_a\rangle
\end{equation*}

For a free particle
\begin{equation*}
\hat H=\frac{\hat p^2}{2m}
\end{equation*}

Hence
\begin{equation*}
K(b,a)=\langle x_b|\exp\left(-\frac{i}{\hbar}\frac{(t_b-t_a)}{2m}\hat p^2\right)|x_a\rangle
\end{equation*}

By the identity
\begin{equation*}
\int_{-\infty}^\infty|p\rangle\langle p|\,dp=1
\end{equation*}

we can write
\begin{equation*}
K(b,a)=\int_{-\infty}^\infty
\langle x_b|\exp\left(-\frac{i}{\hbar}\frac{(t_b-t_a)}{2m}\hat p^2\right)|p\rangle\langle p|x_a\rangle\,dp
\end{equation*}

By the formula $\hat p|p\rangle=p|p\rangle$ we can
substitute eigenvalue $p$ for momentum operator $\hat p$.
\begin{equation*}
K(b,a)=\int\langle x_b|\exp\left(-\frac{i}{\hbar}\frac{(t_b-t_a)}{2m}p^2\right)
|p\rangle\langle p|x_a\rangle\,dp
\end{equation*}

Now that operator $\hat p$ is gone we can rearrange the integrand as
\begin{equation*}
K(b,a)=\int_{-\infty}^\infty
\exp\left(-\frac{i}{\hbar}\frac{(t_b-t_a)}{2m}p^2\right)\langle x_b|p\rangle\langle p|x_a\rangle\,dp
\end{equation*}

For a free particle
\begin{equation*}
\langle x|p\rangle=\frac{1}{\sqrt{2\pi\hbar}}\exp\left(\frac{ipx}{\hbar}\right)
\end{equation*}

Hence
\begin{equation*}
K(b,a)=\int_{-\infty}^\infty
\exp\left(-\frac{i}{\hbar}\frac{(t_b-t_a)}{2m}p^2\right)
\frac{1}{\sqrt{2\pi\hbar}}\exp\left(\frac{ipx_b}{\hbar}\right)
\frac{1}{\sqrt{2\pi\hbar}}\exp\left(-\frac{ipx_a}{\hbar}\right)\,dp
\end{equation*}

Simplify as
\begin{equation*}
K(b,a)=\frac{1}{2m\hbar}\int_{-\infty}^\infty
\exp\left(-\frac{i}{\hbar}\frac{(t_b-t_a)}{2m}p^2+\frac{i}{\hbar}(x_b-x_a)p\right)\,dp
\end{equation*}

Let
\begin{equation*}
a=\frac{i}{\hbar}\frac{(t_b-t_a)}{2m}
\end{equation*}

and
\begin{equation*}
b=\frac{i}{\hbar}(x_b-x_a)
\end{equation*}

so that
\begin{equation*}
K(b,a)=\frac{1}{2m\hbar}
\int_{-\infty}^\infty
\exp\left(-ap^2+bp\right)\,dp
\end{equation*}

By the Gaussian integral formula
\begin{equation*}
\int_{-\infty}^\infty\exp\left(-ap^2+bp\right)\,dp
=\sqrt{\frac{\pi}{a}}\exp\left(\frac{b^2}{4a}\right)
\end{equation*}

we have
\begin{equation*}
K(b,a)=\frac{1}{2\pi\hbar}
\sqrt{\frac{2\pi m\hbar}{i(t_b-t_a)}}
\exp\left(-\frac{m(x_b-x_a)^2}{2i\hbar(t_b-t_a)}\right)
\end{equation*}

Rearrange as
\begin{equation*}
K(b,a)=
\sqrt{\frac{m}{2\pi i\hbar(t_b-t_a)}}
\exp\left(\frac{im(x_b-x_a)^2}{2\hbar(t_b-t_a)}\right)
\tag{1}
\end{equation*}
