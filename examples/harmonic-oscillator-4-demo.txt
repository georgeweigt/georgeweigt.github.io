"Harmonic oscillator 4"

N = 4

a1 = zero(N,N)
a2 = zero(N,N)

for(k,0,N-1,for(j,0,N-1,
 a1[k+1,j+1] = sqrt(j + 1) (k == j + 1),
 a2[k+1,j+1] = sqrt(j) (k == j - 1)
))

X = sqrt(hbar / (2 m omega)) (a1 + a2)
P = i sqrt(m hbar omega / 2) (a1 - a2)

H = dot(P,P) / (2 m) + 1/2 m omega^2 dot(X,X)

H[N,N] = (N - 1 + 1/2) hbar omega -- correction
H

C = 1/sqrt(2) (1,1,0,0)

"Verify equation (1)"
check(dot(conj(C),X,C) == sqrt(hbar / (2 m omega)))
"ok"
