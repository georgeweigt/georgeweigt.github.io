\section*{Maxwell equations}

Let $\mathbf E$ be the electric field
\begin{equation*}
\mathbf E=E_0\sin(kz-\omega t)
\begin{pmatrix}
1\\0\\0
\end{pmatrix}
\end{equation*}

and $\mathbf B$ the magnetic field
\begin{equation*}
\mathbf B=\frac{E_0}{c}\sin(kz-\omega t)
\begin{pmatrix}
0\\1\\0
\end{pmatrix}
\end{equation*}

where $k=\omega/c$.
Verify $\mathbf E$ and $\mathbf B$ are solutions to the free-field Maxwell equations
\begin{align*}
\nabla\cdot\mathbf E&=0
\\
\nabla\cdot\mathbf B&=0
\\
\nabla\times\mathbf E&=-\frac{\partial}{\partial t}{\mathbf B}
\\[1ex]
\nabla\times\mathbf B&=\frac{1}{c^2}\frac{\partial}{\partial t}{\mathbf E}
\end{align*}

Electric field $\mathbf E$ is measured in Newtons per Coulomb.
\begin{equation*}
E_0=\frac{[\text{N}]}{\text{[C]}}
\end{equation*}

Magnetic field $\mathbf B$ is measured in Tesla.
\begin{equation*}
\frac{E_0}{c}=\frac{[\text{N}]}{\text{[C]}}\frac{[\text{s}]}{[\text{m}]}=[\text{T}]
\end{equation*}

Let $\mathbf A$ be the magnetic vector potential
\begin{equation*}
\mathbf A=-\frac{E_0}{\omega}\cos(kz-\omega t)
\begin{pmatrix}
1\\0\\0
\end{pmatrix}
\end{equation*}

Verify
\begin{align*}
\mathbf E&=-\frac{\partial}{\partial t}\mathbf A
\\
\mathbf B&=\nabla\times\mathbf A
\end{align*}

Magnetic vector potential $\mathbf A$ is measured in Tesla meters.
\begin{equation*}
\frac{E_0}{\omega}=\frac{[\text{N}]}{\text{[C]}}[\text{s}]=[\text{T}\,\text{m}]
\end{equation*}
