\section*{Compton scattering 4}

From the first section we have
\begin{align*}
\sum_{ab}\mathcal M_{1}{}^{\mu\nu}\mathcal M^*_{1\mu\nu}
&=\frac{e^4}{(s-m^2)^2}
\left(
-8 s u + 24 m^2 s + 8 m^2 u + 8 m^4
\right)
\\
\sum_{ab}\mathcal M_{1}{}^{\mu\nu}\mathcal M^*_{2\nu\mu}
&=\frac{e^4}{(s-m^2)(u-m^2)}
\left(
8 m^2 s + 8 m^2 u + 16 m^4
\right)
\\
\sum_{ab}\mathcal M_{2}{}^{\nu\mu}\mathcal M^*_{2\nu\mu}
&=\frac{e^4}{(u-m^2)^2}
\left(
-8 s u + 8 m^2 s + 24 m^2 u + 8 m^4
\right)
\end{align*}

We also have by computer algebra
\begin{align*}
\sum_{ab}\mathcal M_{1}{}^{\mu\nu}\mathcal M^*_{1\mu\nu}
&=\frac{e^4}{(s-m^2)^2}
\left(
32 (p_1\cdot p_2) (p_1\cdot p_4) + 32 m^2 (p_1\cdot p_2) + 32 m^4
\right)
\\
\sum_{ab}\mathcal M_{1}{}^{\mu\nu}\mathcal M^*_{2\nu\mu}
&=\frac{e^4}{(s-m^2)(u-m^2)}
\left(
16 m^2 (p_1\cdot p_2) - 16 m^2 (p_1\cdot p_4) + 32 m^4
\right)
\\
\sum_{ab}\mathcal M_{2}{}^{\nu\mu}\mathcal M^*_{2\nu\mu}
&=\frac{e^4}{(u-m^2)^2}
\left(
32 (p_1\cdot p_2) (p_1\cdot p_4) - 32 m^2 (p_1\cdot p_4) + 32 m^4
\right)
\end{align*}
