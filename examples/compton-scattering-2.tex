\section*{Compton scattering 2}

From the previous section we have
\begin{equation*}
p_1=\underset{e^- \, \longrightarrow}
{\begin{pmatrix}E\\0\\0\\ \omega\end{pmatrix}}
\qquad
p_2=\underset{\longleftarrow \, \gamma}
{\begin{pmatrix}\omega\\0\\0\\-\omega\end{pmatrix}}
\qquad
p_3=\underset{\substack{\phantom{\nearrow} \, e^-\\\nearrow \, \phantom{e^-}}}
{\begin{pmatrix}
E\\
\omega\sin\theta\cos\phi\\
\omega\sin\theta\sin\phi\\
\omega\cos\theta
\end{pmatrix}}
\qquad
p_4=\underset{\substack{\phantom{\gamma} \, \swarrow\\\gamma \, \phantom{\swarrow}}}
{\begin{pmatrix}
\omega\\
-\omega\sin\theta\cos\phi\\
-\omega\sin\theta\sin\phi\\
-\omega\cos\theta
\end{pmatrix}}
\end{equation*}

and
\begin{equation*}
\langle|\mathcal{M}|^2\rangle=\frac{e^4}{4}
\left(\frac{f_{11}}{(s-m^2)^2}+\frac{2f_{12}}{(s-m^2)(u-m^2)}+\frac{f_{22}}{(u-m^2)^2}\right)
\end{equation*}

where
\begin{align*}
f_{11}&=-8 s u + 24 m^2 s + 8 m^2 u + 8 m^4
\\
f_{12}&=8 m^2 s + 8 m^2 u + 16 m^4
\\
f_{22}&=-8 s u + 8 m^2 s + 24 m^2 u + 8 m^4
\end{align*}

Compton scattering experiments are typically done in the lab frame where the electron is at rest.
Define boost $\Lambda$ for transforming momentum vectors to the lab frame.
\begin{equation*}
\Lambda=
\begin{pmatrix}
E/m & 0 & 0 & -\omega/m\\
0 & 1 & 0 & 0\\
0 & 0 & 1 & 0\\
\omega/m & 0 & 0 & -E/m
\end{pmatrix}
\end{equation*}

The electron is at rest in the lab frame.
\begin{equation*}
\Lambda p_1=\begin{pmatrix}m\\0\\0\\0\end{pmatrix}
\end{equation*}

Mandelstam variables are invariant under a boost.
\begin{equation*}
\begin{aligned}
s&=(p_1+p_2)^2=(\Lambda p_1+\Lambda p_2)^2
\\
t&=(p_1-p_3)^2=(\Lambda p_1-\Lambda p_3)^2
\\
u&=(p_1-p_4)^2=(\Lambda p_1-\Lambda p_4)^2
\end{aligned}
\end{equation*}

In the lab frame, let $\omega_L$ be the angular frequency of the incident photon
and let $\omega_L'$ be the angular frequency of the scattered photon.
\begin{equation*}
\begin{aligned}
\omega_L&=\Lambda p_2\cdot\begin{pmatrix}1\\0\\0\\0\end{pmatrix}
=\frac{\omega^2}{m}+\frac{\omega E}{m}
\\[1ex]
\omega_L'&=\Lambda p_4\cdot\begin{pmatrix}1\\0\\0\\0\end{pmatrix}
=\frac{\omega^2\cos\theta}{m}+\frac{\omega E}{m}
\end{aligned}
\end{equation*}

By computer algebra we have
\begin{equation*}
\begin{aligned}
s&=m^2+2m\omega_L
\\
t&=2m(\omega_L' - \omega_L)
\\
u&=m^2-2 m \omega_L'
\end{aligned}
\end{equation*}

Hence
\begin{equation*}
\langle|\mathcal{M}|^2\rangle=2e^4
\left[
\frac{\omega_L}{\omega_L'}+\frac{\omega_L'}{\omega_L}
+\left(\frac{m}{\omega_L}-\frac{m}{\omega_L'}+1\right)^2-1
\right]
\end{equation*}

Lab scattering angle $\theta_L$ is given by the Compton equation
\begin{equation*}
\cos\theta_L=\frac{m}{\omega_L}-\frac{m}{\omega_L'}+1
\end{equation*}

Hence
\begin{equation*}
\langle|\mathcal{M}|^2\rangle
=2e^4\left(\frac{\omega_L}{\omega_L'}+\frac{\omega_L'}{\omega_L}+\cos^2\theta_L-1\right)
\end{equation*}

Substitute $-\sin^2\theta_L$ for $\cos^2\theta_L-1$.
\begin{equation*}
\langle|\mathcal{M}|^2\rangle
=2e^4\left(\frac{\omega_L}{\omega_L'}+\frac{\omega_L'}{\omega_L}-\sin^2\theta_L\right)
\end{equation*}

For simplicity let us drop the $L$ subscript from lab variables.
From now on the symbols $\omega$, $\omega'$, and $\theta$ will be lab frame variables.

\bigskip
The differential cross section is
\begin{equation*}
\frac{d\sigma}{d\Omega}
=\frac{\alpha^2}{e^4}
\left(\frac{\omega'}{\omega}\right)^2
\frac{\langle|\mathcal{M}|^2\rangle}{4s}
\end{equation*}

For the lab frame we have
\begin{equation*}
\langle|\mathcal{M}|^2\rangle
=2e^4\left(
\frac{\omega}{\omega'}+\frac{\omega'}{\omega}-\sin^2\theta
\right)
\end{equation*}

Hence
\begin{equation*}
\frac{d\sigma}{d\Omega}
=\frac{\alpha^2}{2s}
\left(\frac{\omega'}{\omega}\right)^2
\left(
\frac{\omega}{\omega'}+\frac{\omega'}{\omega}-\sin^2\theta
\right)
\end{equation*}

\iffalse

Multiply by $(\hbar c)^2$ to convert from natural units to physical units.
\begin{equation*}
\frac{d\sigma}{d\Omega}
=\frac{\alpha^2}{2s}
\left(\frac{\omega'}{\omega}\right)^2
\left(
\frac{\omega}{\omega'}+\frac{\omega'}{\omega}-\sin^2\theta
\right)
(\hbar c)^2
\end{equation*}

\fi

For $\omega\ll m$ we can use the approximation $s=m^2$ and obtain the Klein-Nishina formula
\begin{equation*}
\frac{d\sigma}{d\Omega}
=\frac{\alpha^2}{2m^2}
\left(\frac{\omega'}{\omega}\right)^2
\left(
\frac{\omega}{\omega'}+\frac{\omega'}{\omega}-\sin^2\theta
\right)
\end{equation*}
