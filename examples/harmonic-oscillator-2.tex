\section*{Harmonic oscillator 2}

From the previous section we have
\begin{equation*}
\psi_n=C_n
\exp\left(\frac{m\omega x^2}{2\hbar}\right)
\frac{d^n}{dx^n}\exp\left(-\frac{m\omega x^2}{\hbar}\right)
\end{equation*}

where $C_n$ is the normalization constant
\begin{equation*}
C_n=\frac{(-1)^n}{\sqrt{2^nn!}}
\left(\frac{m\omega}{\pi\hbar}\right)^{1/4}
\left(\frac{\hbar}{m\omega}\right)^{n/2}
\end{equation*}

Evaluate the derivative of $\psi_n$. By the product rule
\begin{equation*}
\frac{d}{dx}\psi_n
=\frac{m\omega x}{\hbar}\psi_n
+C_n
\exp\left(\frac{m\omega x^2}{2\hbar}\right)
\frac{d^{n+1}}{dx^{n+1}}\exp\left(-\frac{m\omega x^2}{\hbar}\right)
\end{equation*}

Substitute $\psi_{n+1}$ for the right-hand side.
\begin{equation*}
\frac{d}{dx}\psi_n
=\frac{m\omega x}{\hbar}\psi_n
+\frac{C_n}{C_{n+1}}\psi_{n+1}
\end{equation*}

We have
\begin{equation*}
\frac{C_n}{C_{n+1}}
%=\frac{\sqrt{2(n+1)}}{(-1)}\left(\frac{\hbar}{m\omega}\right)^{-1/2}
=-\sqrt{\frac{2m\omega}{\hbar}}\sqrt{n+1}
\end{equation*}

Hence
\begin{equation*}
\frac{d}{dx}\psi_n
=\frac{m\omega x}{\hbar}\psi_n
-\sqrt{\frac{2m\omega}{\hbar}}\sqrt{n+1}\psi_{n+1}
\end{equation*}

Rearrange as
\begin{equation*}
\sqrt{\frac{\hbar}{2m\omega}}
\left(\frac{m\omega x}{\hbar}-\frac{d}{dx}\right)\psi_n
=\sqrt{n+1}\psi_{n+1}
\end{equation*}

Let $a_+$ be the operator
\begin{equation*}
a_+=\sqrt{\frac{\hbar}{2m\omega}}
\left(\frac{m\omega x}{\hbar}-\frac{d}{dx}\right)
\end{equation*}

Hence $a_+$ is the step-up operator
\begin{equation*}
a_+\psi_n=\sqrt{n+1}\psi_{n+1}
\end{equation*}

Let $a_-$ be the operator (needs explanation)
\begin{equation*}
a_-=\sqrt{\frac{\hbar}{2m\omega}}
\left(\frac{m\omega x}{\hbar}+\frac{d}{dx}\right)
\end{equation*}

Hence $a_-$ is the step-down operator
\begin{equation*}
a_-\psi_n=\sqrt{n}\psi_{n-1}
\end{equation*}

Let $N$ be the number operator
\begin{equation*}
N=a_+a_-
\end{equation*}

Hence
\begin{equation*}
N\psi_n=a_+a_-\psi_n=a_+\left(\sqrt n\psi_{n-1}\right)=n\psi_n
\end{equation*}
