\section*{Annihilation}

Annihilation is the process $e^-+e^+\rightarrow\gamma+\gamma$.
The arrow for $e^+$ is reversed because a positron is treated
mathematically as an electron moving backwards in time.

\begin{center}
\begin{tikzpicture}
\draw[dashed] (0,0) circle (0.5cm);
\draw[thick,<-] (2,0) node[anchor=west] {$e^+$} -- (0.6,0);
\draw[thick,->] (-2,0) node[anchor=east] {$e^-$} -- (-0.6,0);
\draw[thick,->] (0.40,0.40) -- (1.3,1.3) node[anchor=south west] {$\gamma$};
\draw[thick,->] (-0.4,-0.4) -- (-1.3,-1.3) node[anchor=north east] {$\gamma$};
\draw (1,0.5) node {$\theta$};
\end{tikzpicture}
\end{center}

Momentum vectors where $E=\sqrt{p^2+m^2}$.
\begin{equation*}
p_1=\underset{e^- \, \longrightarrow}
{\begin{pmatrix}E\\0\\0\\p\end{pmatrix}}
\qquad
p_2=\underset{\longrightarrow \, e^+}
{\begin{pmatrix}E\\0\\0\\-p\end{pmatrix}}
\qquad
p_3=\underset{\substack{\phantom{\nearrow} \, \gamma\\\nearrow \, \phantom{\gamma}}}
{\begin{pmatrix}E\\ E\sin\theta\cos\phi\\ E\sin\theta\sin\phi\\ E\cos\theta\end{pmatrix}}
\qquad
p_4=\underset{\substack{\phantom{\gamma} \, \swarrow\\\gamma \, \phantom{\swarrow}}}
{\begin{pmatrix}E\\ -E\sin\theta\cos\phi\\ -E\sin\theta\sin\phi\\ -E\cos\theta\end{pmatrix}}
\end{equation*}

Spinors for $p_1$.
\begin{equation*}
u_{11}=\frac{1}{\sqrt{E+m}}
\underset{\text{spin up}}
{\begin{pmatrix}E+m\\0\\p\\0\end{pmatrix}}
\qquad
u_{12}=\frac{1}{\sqrt{E+m}}
\underset{\text{spin down}}
{\begin{pmatrix}0\\E+m\\0\\-p\end{pmatrix}}
\end{equation*}

Spinors for $p_2$.
\begin{equation*}
v_{21}=\frac{1}{\sqrt{E+m}}
\underset{\text{spin up}}
{\begin{pmatrix}-p\\0\\E+m\\0\end{pmatrix}}
\qquad
v_{22}=\frac{1}{\sqrt{E+m}}
\underset{\text{spin down}}
{\begin{pmatrix}0\\p\\0\\E+m\end{pmatrix}}
\end{equation*}

The scattering amplitude $\mathcal M$ is
\begin{equation*}
\mathcal M=\mathcal M_1+\mathcal M_2
\end{equation*}

where $\mathcal M_1$ is the $t$-channel amplitude
\begin{equation*}
\mathcal M_1=\frac{\bar v_{2b}(-ie\gamma^\mu)(\slashed q_1+m)(-ie\gamma^\nu)u_{1a}}{t-m^2}
\end{equation*}

and $\mathcal M_2$ is the $u$-channel amplitude
\begin{equation*}
\mathcal M_2=\frac{\bar v_{2b}(-ie\gamma^\nu)(\slashed q_2+m)(-ie\gamma^\mu)u_{1a}}{u-m^2}
\end{equation*}

Matrices $\slashed q_1$ and $\slashed q_2$ are momentum transfer.
\begin{align*}
\slashed q_1&=(p_1-p_3)^\alpha g_{\alpha\beta}\gamma^\beta
\\
\slashed q_2&=(p_1-p_4)^\alpha g_{\alpha\beta}\gamma^\beta
\end{align*}

Expected density $\langle|\mathcal M|^2\rangle$ is the average density.
\begin{equation*}
\langle|\mathcal M|^2\rangle
=\frac{1}{4}\sum_{ab}|\mathcal M|^2
\end{equation*}

Expand the summand.
(Note indices $\mu$ and $\nu$ of $\mathcal M_2$ are interchanged relative to $\mathcal M_1$.)
\begin{equation*}
\langle|\mathcal{M}|^2\rangle
=\frac{1}{4}\sum_{ab}
\left(
\mathcal M_{1}{}^{\mu\nu}\mathcal M^*_{1\mu\nu}
+\mathcal M_{1}{}^{\mu\nu}\mathcal M^*_{2\nu\mu}
+\mathcal M_{2}{}^{\nu\mu}\mathcal M^*_{1\mu\nu}
+\mathcal M_{2}{}^{\nu\mu}\mathcal M^*_{2\nu\mu}
\right)
\end{equation*}

The summation results are
\begin{align*}
\sum_{ab}\mathcal M_1{}^{\mu\nu}\mathcal M^*_{1\mu\nu}
&=\frac{e^4}{(t-m^2)^2}
\left(8 t u - 24 m^2 t - 8 m^2 u - 8 m^4\right)
\\
\sum_{ab}\mathcal M_1{}^{\mu\nu}\mathcal M^*_{1\nu\mu}
&=\frac{e^4}{(t-m^2)(u-m^2)}
\left(8 m^2 s - 32 m^4\right)
\\
\sum_{ab}\mathcal M_2{}^{\nu\mu}\mathcal M^*_{2\nu\mu}
&=\frac{e^4}{(u-m^2)^2}
\left(8 t u - 8 m^2 t - 24 m^2 u - 8 m^4\right)
\end{align*}

where $s$, $t$, and $u$ are Mandelstam variables
\begin{align*}
s&=(p_1+p_2)^2
\\
t&=(p_1-p_3)^2
\\
u&=(p_1-p_4)^2
\end{align*}

For clarity define
\begin{align*}
f_{11}&=8 t u - 24 m^2 t - 8 m^2 u - 8 m^4
\\
f_{12}&=8 m^2 s - 32 m^4
\\
f_{22}&=8 t u - 8 m^2 t - 24 m^2 u - 8 m^4
\end{align*}

and write $\langle|\mathcal{M}|^2\rangle$ more compactly as
\begin{equation*}
\langle|\mathcal{M}|^2\rangle=\frac{e^4}{4}
\left(
\frac{f_{11}}{(t-m^2)^2}+\frac{2f_{12}}{(t-m^2)(u-m^2)}+\frac{f_{22}}{(u-m^2)^2}
\right)
\end{equation*}

For $E\gg m$ a useful approximation is to set $m=0$ and obtain
\begin{align*}
f_{11}&=8tu
\\
f_{12}&=0
\\
f_{22}&=8tu
\end{align*}

Then $\langle|\mathcal{M}|^2\rangle$ simplifies as
\begin{equation*}
\langle|\mathcal{M}|^2\rangle
=\frac{e^4}{4}\left(\frac{8tu}{t^2}+\frac{8tu}{u^2}\right)
=2e^4\left(\frac{u}{t}+\frac{t}{u}\right)
\end{equation*}

For $m=0$ the Mandelstam variables are
\begin{align*}
s&=4E^2
\\
t&=-2E^2(1-\cos\theta)
\\
u&=-2E^2(1+\cos\theta)
\end{align*}

Hence
\begin{equation*}
\langle|\mathcal{M}|^2\rangle
=2e^4\left(\frac{1+\cos\theta}{1-\cos\theta}+\frac{1-\cos\theta}{1+\cos\theta}\right)
\end{equation*}

The differential cross section is
\begin{equation*}
\frac{d\sigma}{d\Omega}
=\frac{\alpha^2}{e^4}\frac{\langle|\mathcal{M}|^2\rangle}{4s}
=\frac{\alpha^2}{8E^2}
\left(\frac{1+\cos\theta}{1-\cos\theta}+\frac{1-\cos\theta}{1+\cos\theta}\right)
\end{equation*}

Multiply by $(\hbar c)^2$ to convert from natural units to physical units.
\begin{equation*}
\frac{d\sigma}{d\Omega}
=\frac{\alpha^2}{8E^2}
\left(\frac{1+\cos\theta}{1-\cos\theta}+\frac{1-\cos\theta}{1+\cos\theta}\right)
(\hbar c)^2
\end{equation*}
