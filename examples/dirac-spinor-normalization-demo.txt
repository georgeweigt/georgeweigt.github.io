"Dirac spinor normalization"

E = sqrt(px^2 + py^2 + pz^2 + m^2)

"Verify equation (1)"
u = sqrt(E + m) (1, 0, pz / (E + m), (px + i py) / (E + m))
check(dot(conj(u),u) == 2 E)
"ok"

"Verify equation (2)"
u = (1, 0, pz / (E + m), (px + i py) / (E + m))
check(dot(conj(u),u) == (px^2 + py^2 + pz^2) / (E + m)^2 + 1)
"ok"
