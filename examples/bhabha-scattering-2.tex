\section*{Bhabha scattering 2}

For $t$-channel scattering amplitudes $\mathcal M_t$ we have
\begin{equation*}
\mathcal M_t=\iint
\bar\psi_2(y)
(-ie\gamma^\mu)
\psi_4(y)
G(x,y)
\bar\psi_3(x)
(-ie\gamma_\mu)
\psi_1(x)
\,d^4x\,d^4y
\end{equation*}

where $G(x,y)$ is the photon propagator
\begin{equation*}
G(x,y)=\frac{1}{(2\pi)^4}\int\frac{\exp(-ip\cdot(x-y))}{p^2+i\epsilon}\,d^4p
\end{equation*}

The wave functions are
\begin{align*}
\psi_1(x)&=\sqrt{2EV}u_1\exp(-ip_1\cdot x)
\\
\psi_2(x)&=\sqrt{2EV}v_2\exp(ip_2\cdot x)
\\
\psi_3(x)&=\sqrt{2EV}u_3\exp(-ip_3\cdot x)
\\
\psi_4(x)&=\sqrt{2EV}v_4\exp(ip_4\cdot x)
\end{align*}

where $V$ is a normalization volume.

\bigskip
Substitute for the wave functions in $\mathcal M_t$ and factor the spinor products.
\begin{equation*}
\mathcal M_t=-\frac{e^2}{(2EV)^2}(\bar v_2\gamma^\mu v_4)(\bar u_3\gamma_\mu u_1)
\iint
{\exp(i(-p_2+p_4)\cdot y)}
\,G(x,y)\,
{\exp(i(p_3-p_1)\cdot x)}
\,d^4x\,d^4y
\end{equation*}

Substitute for $G(x,y)$.
\begin{multline*}
\mathcal M_t=-\frac{e^2}{(2EV)^2}(\bar v_2\gamma^\mu v_4)(\bar u_3\gamma_\mu u_1)
\\
\frac{1}{(2\pi)^4}
\iiint
{\exp(i(-p_2+p_4)\cdot y)}
\,{\frac{\exp(-ip\cdot(x-y))}{p^2+i\epsilon}}\,
{\exp(i(p_3-p_1)\cdot x)}
\,d^4x\,d^4y\,d^4p
\end{multline*}

Combine exponentials.
\begin{multline*}
\mathcal M_t=-\frac{e^2}{(2EV)^2}(\bar v_2\gamma^\mu v_4)(\bar u_3\gamma_\mu u_1)
\\
\frac{1}{(2\pi)^4}
\iiint
\frac{\exp(i(-p_2+p_4+p)\cdot y)\exp(i(p_3-p_1-p)\cdot x)}{p^2+i\epsilon}
\,d^4x\,d^4y\,d^4p
\end{multline*}

Solve the integral over $x$.
\begin{equation*}
\int\exp(i(p_3-p_1-p)\cdot x)\,d^4x=(2\pi)^4\delta^4(p_3-p_1-p)
\end{equation*}

Solve the integral over $y$.
\begin{equation*}
\int\exp(i(-p_2+p_4+p)\cdot y)\,d^4y=(2\pi)^4\delta^4(-p_2+p_4+p)
\end{equation*}

Hence
\begin{equation*}
\mathcal M_t=-\frac{e^2}{(2EV)^2}(\bar v_2\gamma^\mu v_4)(\bar u_3\gamma_\mu u_1)
(2\pi)^4\int
\frac{\delta^4(-p_2+p_4+p)\delta^4(p_3-p_1-p)}{p^2+i\epsilon}
\,d^4p
\end{equation*}

The integral over $p$ vanishes except for
\begin{equation*}
p=p_2-p_4=p_3-p_1
\end{equation*}

Hence
\begin{equation*}
\mathcal M_t=-\frac{e^2}{(2EV)^2}(\bar v_2\gamma^\mu v_4)(\bar u_3\gamma_\mu u_1)
(2\pi)^4\,\frac{\delta^4(p_4+p_3-p_2-p_1)}{(p_3-p_1)^2}
\end{equation*}

For momentum vectors
\begin{equation*}
p_1=\underset{e^- \, \longrightarrow}
{\begin{pmatrix}E\\0\\0\\p\end{pmatrix}}
\qquad
p_2=\underset{\longleftarrow \, e^+}
{\begin{pmatrix}E\\0\\0\\-p\end{pmatrix}}
\qquad
p_3=\underset{\substack{\phantom{e^-} \, \nearrow\\e^- \, \phantom{\nearrow}}}
{\begin{pmatrix}
E\\
p\sin\theta\cos\phi\\
p\sin\theta\sin\phi\\
p\cos\theta
\end{pmatrix}}
\qquad
p_4=\underset{\substack{\phantom{\swarrow} \, e^+\\\swarrow \, \phantom{e^+}}}
{\begin{pmatrix}
E\\
-p\sin\theta\cos\phi\\
-p\sin\theta\sin\phi\\
-p\cos\theta
\end{pmatrix}}
\end{equation*}

we have $\delta^4(p_4+p_3-p_2-p_1)=\delta^4(0)=1$ hence
\begin{equation*}
\mathcal M_t=-\frac{e^2}{(2EV)^2}(\bar v_2\gamma^\mu v_4)(\bar u_3\gamma_\mu u_1)
\frac{(2\pi)^4}{t}
\end{equation*}

where $t$ is the Mandelstam variable
\begin{equation*}
t=(p_1-p_3)^2
\end{equation*}

The $s$-channel scattering amplitudes $\mathcal M_s$
have $\psi_2$ and $\psi_3$ interchanged relative to $\mathcal M_t$.
\begin{equation*}
\mathcal M_t=\iint
\bar\psi_3(y)
(-ie\gamma^\mu)
\psi_4(y)
G(x,y)
\bar\psi_2(x)
(-ie\gamma_\mu)
\psi_1(x)
\,d^4x\,d^4y
\end{equation*}

It follows that
\begin{equation*}
\mathcal M_t=-\frac{e^2}{(2EV)^2}(\bar u_3\gamma^\mu v_4)(\bar v_2\gamma_\mu u_1)
(2\pi)^4\int
\frac{\delta^4(p_3+p_4+p)\delta^4(-p_2-p_1-p)}{p^2+i\epsilon}
\,d^4p
\end{equation*}

The integral over $p$ vanishes except for
\begin{equation*}
p=-p_3-p_4=p_1+p_2
\end{equation*}

Hence
\begin{equation*}
\mathcal M_s=-\frac{e^2}{(2EV)^2}(\bar u_3\gamma^\mu v_4)(\bar v_2\gamma_\mu u_1)
\frac{(2\pi)^4}{s}
\end{equation*}

where $s$ is the Mandelstam variable
\begin{equation*}
s=(p_1+p_2)^2
\end{equation*}
