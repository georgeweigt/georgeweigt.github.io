\section*{Harmonic oscillator 3}

From the previous section
\begin{align*}
a_+&=\sqrt{\frac{\hbar}{2m\omega}}
\left(\frac{m\omega x}{\hbar}-\frac{d}{dx}\right)
\\[1ex]
a_-&=\sqrt{\frac{\hbar}{2m\omega}}
\left(\frac{m\omega x}{\hbar}+\frac{d}{dx}\right)
\end{align*}

Substitute $ip/\hbar$ for $d/dx$.
\begin{align*}
a_+&=\sqrt{\frac{\hbar}{2m\omega}}
\left(\frac{m\omega x}{\hbar}-\frac{ip}{\hbar}\right)
\\[1ex]
a_-&=\sqrt{\frac{\hbar}{2m\omega}}
\left(\frac{m\omega x}{\hbar}+\frac{ip}{\hbar}\right)
\end{align*}

Factor $\hbar$.
\begin{align*}
a_+&=\frac{1}{\sqrt{2m\hbar\omega}}(m\omega x-ip)
\\[1ex]
a_-&=\frac{1}{\sqrt{2m\hbar\omega}}(m\omega x+ip)
\end{align*}

Separate $x$ and $p$.
\begin{align*}
a_++a_-&=\frac{2}{\sqrt{2m\hbar\omega}}m\omega x
\\
a_+-a_-&=-\frac{2}{\sqrt{2m\hbar\omega}}ip
\end{align*}

Solve for $x$ and $p$.
\begin{align*}
x&=\sqrt{\frac{\hbar}{2m\omega}}(a_++a_-)\tag{1}
\\[1ex]
p&=i\sqrt{\frac{m\hbar\omega}{2}}(a_+-a_-)\tag{2}
\end{align*}

From the previous section
\begin{align*}
a_+\psi_j&=\sqrt{j+1}\psi_{j+1}
\\
a_-\psi_j&=\sqrt j\psi_{j-1}
\end{align*}

Hence
\begin{align*}
x\psi_j&=\sqrt{\frac{\hbar}{2m\omega}}\left(\sqrt{j+1}\psi_{j+1}+\sqrt j\psi_{j-1}\right)
\\[1ex]
p\psi_j&=i\sqrt{\frac{m\hbar\omega}{2}}\left(\sqrt{j+1}\psi_{j+1}-\sqrt j\psi_{j-1}\right)
\end{align*}
