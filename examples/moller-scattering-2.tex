\section*{Moller scattering 2}

For transition amplitudes $M_t$ we have
\begin{equation*}
M_t=\iint
\bar\psi_4(y)\underset{\text{vertex}}{(-ie\gamma^\mu)}\psi_2(y)
\underset{\substack{\text{photon}\\\text{propagator}}}{G(x,y)}
\bar\psi_3(x)\underset{\text{vertex}}{(-ie\gamma_\mu)}\psi_1(x)
\,d^4x\,d^4y
\end{equation*}

where $G(x,y)$ is the photon propagator
\begin{equation*}
G(x,y)=\frac{1}{(2\pi)^4}\int\frac{\exp(-ip(x-y))}{p^2+i\epsilon}\,d^4p
\end{equation*}

Moller scattering wave functions normalized as $\langle\psi|\psi\rangle=1$ over volume $V$.
\begin{align*}
\psi_1(x^\mu)=\sqrt{2EV}u_1\exp(-ip_1^\mu g_{\mu\nu}x^\nu)
\\
\psi_2(x^\mu)=\sqrt{2EV}u_2\exp(-ip_2^\mu g_{\mu\nu}x^\nu)
\\
\psi_3(x^\mu)=\sqrt{2EV}u_3\exp(-ip_3^\mu g_{\mu\nu}x^\nu)
\\
\psi_4(x^\mu)=\sqrt{2EV}u_4\exp(-ip_4^\mu g_{\mu\nu}x^\nu)
\end{align*}

For clarity we abbreviate as
\begin{align*}
\psi_1(x)=\sqrt{2EV}u_1\exp(-ip_1x)
\\
\psi_2(x)=\sqrt{2EV}u_2\exp(-ip_2x)
\\
\psi_3(x)=\sqrt{2EV}u_3\exp(-ip_3x)
\\
\psi_4(x)=\sqrt{2EV}u_4\exp(-ip_4x)
\end{align*}

Substitute for the wave functions in $M_t$ and factor the spinors.
\begin{multline*}
M_t=-\frac{e^2}{(2EV)^2}(\bar u_4\gamma^\mu u_2)(\bar u_3\gamma_\mu u_1)
\\
{}\times
\iint
\exp(ip_4y)\exp(-ip_2y)
G(x,y)
\exp(ip_3x)\exp(-ip_1x)
\,d^4x\,d^4y
\end{multline*}

Substitute for $G(x,y)$.
\begin{multline*}
M_t=-\frac{e^2}{(2EV)^2}(\bar u_4\gamma^\mu u_2)(\bar u_3\gamma_\mu u_1)
\\
{}\times\frac{1}{(2\pi)^4}
\iiint
\exp(ip_4y)\exp(-ip_2y)
{\frac{\exp(-ip(x-y))}{p^2+i\epsilon}}
\exp(ip_3x)\exp(-ip_1x)
\,d^4x\,d^4y\,d^4p
\end{multline*}

Combine exponentials.
\begin{multline*}
M_t=-\frac{e^2}{(2EV)^2}(\bar u_4\gamma^\mu u_2)(\bar u_3\gamma_\mu u_1)
\\
{}\times\frac{1}{(2\pi)^4}
\iiint
\frac{\exp(i(p_4-p_2+p)y)\exp(i(p_3-p_1-p)x)}{p^2+i\epsilon}
\,d^4x\,d^4y\,d^4p
\end{multline*}

The integrals over $x$ and $y$ are solved as
\begin{align*}
\int\exp(i(p_3-p_1-p)x)\,d^4x=(2\pi)^4\delta^4(p_3-p_1-p)
\\
\int\exp(i(p_4-p_2+p)y)\,d^4y=(2\pi)^4\delta^4(p_4-p_2+p)
\end{align*}

Hence
\begin{equation*}
M_t=-\frac{e^2}{(2EV)^2}(\bar u_4\gamma^\mu u_2)(\bar u_3\gamma_\mu u_1)
(2\pi)^4\int
\frac{\delta^4(p_4-p_2+p)\delta^4(p_3-p_1-p)}{p^2+i\epsilon}
\,d^4p
\end{equation*}

The integral over $p$ vanishes except for
\begin{equation*}
p=p_3-p_1=p_2-p_4
\end{equation*}

Hence
\begin{equation*}
M_t=-\frac{e^2}{(2EV)^2}(\bar u_4\gamma^\mu u_2)(\bar u_3\gamma_\mu u_1)
(2\pi)^4\frac{\delta^4(p_4+p_3-p_2-p_1)}{(p_3-p_1)^2}
\end{equation*}
