\section*{Moller scattering 2}

Here we solve path integrals to find $\mathcal M_1$ and $\mathcal M_2$
that were used in the previous section.

\bigskip
$\mathcal M_1$ is the amplitude for $1\rightarrow3$, $2\rightarrow4$.
\begin{equation*}
\mathcal M_1=i\iint
\bar\psi_4(y)
(-ie\gamma^\mu)
\psi_2(y)
G(x,y)
\bar\psi_3(x)
(-ie\gamma_\mu)
\psi_1(x)
\,d^4x\,d^4y
\end{equation*}

Function $G(x,y)$ is the photon propagator
\begin{equation*}
G(x,y)=\frac{i}{(2\pi)^4}\int\frac{\exp(-ip\cdot(x-y))}{p^2+i\epsilon}\,d^4p
\end{equation*}

The normalized wave functions are ($V$ is volume)
\begin{align*}
\psi_1(x)&=\sqrt{2EV}u_1\exp(-ip_1\cdot x)
\\
\psi_2(x)&=\sqrt{2EV}u_2\exp(-ip_2\cdot x)
\\
\psi_3(x)&=\sqrt{2EV}u_3\exp(-ip_3\cdot x)
\\
\psi_4(x)&=\sqrt{2EV}u_4\exp(-ip_4\cdot x)
\end{align*}

Substitute for the wave functions in $\mathcal M_1$ and factor the spinor products.
\begin{multline*}
\mathcal M_1=-\frac{e^2}{4E^2V^2}(\bar u_4\gamma^\mu u_2)(\bar u_3\gamma_\mu u_1)
\\
i \iint
{\exp(i(p_4-p_2)\cdot y)}
\,G(x,y)\,
{\exp(i(p_3-p_1)\cdot x)}
\,d^4x\,d^4y
\end{multline*}

Substitute for $G(x,y)$.
\begin{multline*}
\mathcal M_1=\frac{e^2}{4E^2V^2}(\bar u_4\gamma^\mu u_2)(\bar u_3\gamma_\mu u_1)
\frac{1}{(2\pi)^4}
\\
\iiint
{\exp(i(p_4-p_2)\cdot y)}
\,{\frac{\exp(-ip\cdot(x-y))}{p^2+i\epsilon}}\,
{\exp(i(p_3-p_1)\cdot x)}
\,d^4x\,d^4y\,d^4p
\end{multline*}

Combine exponentials.
\begin{multline*}
\mathcal M_1=\frac{e^2}{4E^2V^2}(\bar u_4\gamma^\mu u_2)(\bar u_3\gamma_\mu u_1)
\frac{1}{(2\pi)^4}
\\
\iiint
\frac{\exp(i(p_4-p_2+p)\cdot y)\exp(i(p_3-p_1-p)\cdot x)}{p^2+i\epsilon}
\,d^4x\,d^4y\,d^4p
\end{multline*}

Solve the integral over $x$.
\begin{equation*}
\int\exp(i(p_3-p_1-p)\cdot x)\,d^4x=(2\pi)^4\delta^4(p_3-p_1-p)
\end{equation*}

Solve the integral over $y$.
\begin{equation*}
\int\exp(i(p_4-p_2+p)\cdot y)\,d^4y=(2\pi)^4\delta^4(p_4-p_2+p)
\end{equation*}

Hence
\begin{equation*}
\mathcal M_1=\frac{e^2}{4E^2V^2}(\bar u_4\gamma^\mu u_2)(\bar u_3\gamma_\mu u_1)
(2\pi)^4\int
\frac{\delta^4(p_4-p_2+p)\delta^4(p_3-p_1-p)}{p^2+i\epsilon}
\,d^4p
\end{equation*}

The integrand vanishes except for
\begin{equation*}
p=-p_4+p_2=p_3-p_1
\end{equation*}

Hence
\begin{equation*}
\mathcal M_1=\frac{e^2}{4E^2V^2}(\bar u_4\gamma^\mu u_2)(\bar u_3\gamma_\mu u_1)
(2\pi)^4\frac{\delta^4(p_4+p_3-p_2-p_1)}{(p_3-p_1)^2}
\end{equation*}

$\mathcal M_2$ is the amplitude for $1\rightarrow4$, $2\rightarrow3$.
\begin{equation*}
\mathcal M_2=i\iint
\bar\psi_3(y)(-ie\gamma^\mu)\psi_2(y)
G(x,y)
\bar\psi_4(x)(-ie\gamma_\mu)\psi_1(x)
\,d^4x\,d^4y
\end{equation*}

It follows that
\begin{equation*}
\mathcal M_2=\frac{e^2}{4E^2V^2}(\bar u_3\gamma^\mu u_2)(\bar u_4\gamma_\mu u_1)
(2\pi)^4\int
\frac{\delta^4(p_3-p_2+p)\delta^4(p_4-p_1-p)}{p^2+i\epsilon}
\,d^4p
\end{equation*}

The integrand vanishes except for
\begin{equation*}
p=-p_3+p_2=p_4-p_1
\end{equation*}

Hence
\begin{equation*}
\mathcal M_2=\frac{e^2}{4E^2V^2}(\bar u_3\gamma^\mu u_2)(\bar u_4\gamma_\mu u_1)
(2\pi)^4\frac{\delta^4(p_4+p_3-p_2-p_1)}{(p_4-p_1)^2}
\end{equation*}

For the momentum vectors given in the previous section we have
\begin{equation*}
\delta^4(p_4+p_3-p_2-p_1)=\delta^4(0)=1
\end{equation*}

Hence
\begin{align*}
\mathcal M_1&=\frac{(2\pi)^4}{4E^2V^2}
\frac{e^2}{(p_3-p_1)^2}
(\bar u_4\gamma^\mu u_2)(\bar u_3\gamma_\mu u_1)
\\[1ex]
\mathcal M_2&=\frac{(2\pi)^4}{4E^2V^2}
\frac{e^2}{(p_4-p_1)^2}
(\bar u_3\gamma^\mu u_2)(\bar u_4\gamma_\mu u_1)
\end{align*}

In practice the leading coefficient is absorbed in the calculation of $d\sigma$
hence we use the formulas
\begin{align*}
\mathcal M_1&=
\frac{e^2}{(p_3-p_1)^2}
(\bar u_4\gamma^\mu u_2)(\bar u_3\gamma_\mu u_1)
\\[1ex]
\mathcal M_2&=
\frac{e^2}{(p_4-p_1)^2}
(\bar u_3\gamma^\mu u_2)(\bar u_4\gamma_\mu u_1)
\end{align*}
