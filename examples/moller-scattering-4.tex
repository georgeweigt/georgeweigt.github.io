\section*{Moller scattering 4}

From the first section we have
\begin{align*}
\sum_{abcd}\mathcal M_1\mathcal M_1^*&=\frac{e^4}{t^2}
\left(8 s^2 + 8 u^2 - 64 m^2 s - 64 m^2 u + 192 m^4\right)
\\
\sum_{abcd}\mathcal M_1\mathcal M_2^*&=\frac{e^4}{tu}
\left(-8 s^2 + 64 m^2 s - 96 m^4\right)
\\
\sum_{abcd}\mathcal M_2\mathcal M_2^*&=\frac{e^4}{u^2}
\left(8 s^2 + 8 t^2 - 64 m^2 s - 64 m^2 t + 192 m^4\right)
\end{align*}

We also have by computer algebra
\begin{align*}
\sum_{abcd}\mathcal M_1\mathcal M_1^*&=\frac{e^4}{t^2}
\left(32 (p_1\cdot p_2)^2 + 32 (p_1\cdot p_4)^2 - 64 m^2 (p_1\cdot p_2) + 64 m^2 (p_1\cdot p_4)\right)
\\
\sum_{abcd}\mathcal M_1\mathcal M_2^*&=\frac{e^4}{tu}
\left(-32 (p_1\cdot p_2)^2 + 64 m^2 (p_1\cdot p_2)\right)
\\
\sum_{abcd}\mathcal M_2\mathcal M_2^*&=\frac{e^4}{u^2}
\left(32 (p_1\cdot p_2)^2 + 32 (p_1\cdot p_3)^2 - 64 m^2 (p_1\cdot p_2) + 64 m^2 (p_1\cdot p_3)\right)
\end{align*}

For $m=0$
\begin{align*}
32(p_1\cdot p_2)^2&=8 s^2
\\
32(p_1\cdot p_3)^2&=8 t^2
\\
32(p_1\cdot p_4)^2&=8 u^2
\end{align*}
